# ------------------------------------------------------------
# EKS Terraform Setup (GPU telemetry demo cluster)
# ------------------------------------------------------------

# 1) Install prerequisites
# Terraform: https://developer.hashicorp.com/terraform/install
# kubectl:   https://kubernetes.io/docs/tasks/tools/
# AWS CLI v2:
# https://docs.aws.amazon.com/cli/latest/userguide/getting-started-install.html

# 2) Configure AWS credentials (choose one)
aws configure
# or for SSO profiles:
# aws configure sso

# 3) Verify caller identity and default region
aws sts get-caller-identity
aws configure get region

# Optional: override region in the shell for this session
export AWS_REGION=us-west-2

# ------------------------------------------------------------
# GPU quota preparation for NVIDIA L4 Spot (g6 family)
# ------------------------------------------------------------
# L4 on AWS is delivered via g6 instance types. For Spot-backed GPU workers,
# request an EC2 quota increase for G/VT Spot usage before setting
# gpu_node_desired_count > 0.

aws service-quotas list-service-quotas \
  --service-code ec2 \
  --region ${AWS_REGION:-us-west-2} \
  --query 'Quotas[?contains(QuotaName, `Spot`) || contains(QuotaName, `G and VT`) || contains(QuotaName, `Running On-Demand G`)][QuotaName,QuotaCode,Value]' \
  --output table

# Request increase for "All G and VT Spot Instance Requests".
SPOT_QUOTA_CODE=$(aws service-quotas list-service-quotas \
  --service-code ec2 \
  --region ${AWS_REGION:-us-west-2} \
  --query 'Quotas[?contains(QuotaName, `All G and VT Spot Instance Requests`) == `true`].QuotaCode | [0]' \
  --output text)

aws service-quotas request-service-quota-increase \
  --service-code ec2 \
  --quota-code "$SPOT_QUOTA_CODE" \
  --desired-value 16 \
  --region ${AWS_REGION:-us-west-2}

# If your workload can fall back to On-Demand g6, request that quota too.

# ------------------------------------------------------------
# Terraform workflow
# ------------------------------------------------------------

# 4) Copy variables and edit values
cp terraform/eks/terraform.tfvars.example terraform/eks/terraform.tfvars

# 5) Init / plan / apply the cluster root
terraform -chdir=terraform/eks init -input=false
terraform -chdir=terraform/eks plan --var-file=terraform.tfvars
terraform -chdir=terraform/eks apply --var-file=terraform.tfvars

# 6) Configure kubectl for the new cluster
aws eks update-kubeconfig \
  --name gpu-telemetry-demo \
  --region ${AWS_REGION:-us-west-2}

# 7) Install the telemetry stack + demo workloads (in-cluster root)
cp terraform/incluster/terraform.tfvars.example terraform/incluster/terraform.tfvars
#    Set cloud = "eks" (the root authenticates via the current kubeconfig context).

terraform -chdir=terraform/incluster init -input=false
terraform -chdir=terraform/incluster apply --var-file=terraform.tfvars

# 8) Verify
kubectl get nodes -l workload=gpu-telemetry   # GPU node group
kubectl get pods -n gpu-telemetry             # DCGM exporter, gadget, Prometheus, Grafana

# Notes:
# - The GPU node group is tainted nvidia.com/gpu=present:NoSchedule; DCGM and
#   the demo workloads carry matching tolerations (values-eks.yaml).
# - AL2023_x86_64_NVIDIA ships the driver; values-eks.yaml runs the GPU
#   Operator with driver.enabled=false so it only adds toolkit + device plugin.

# ------------------------------------------------------------
# Cleanup
# ------------------------------------------------------------
terraform -chdir=terraform/incluster destroy --var-file=terraform.tfvars
terraform -chdir=terraform/eks destroy --var-file=terraform.tfvars
